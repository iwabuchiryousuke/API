DISCOVER = r"""() => {
  const actions = [...document.querySelectorAll('button,[role="button"],a')];
  const byCard = new Map();
  const now = new Date();
  const today = `${now.getFullYear()}/${String(now.getMonth()+1).padStart(2,'0')}/${String(now.getDate()).padStart(2,'0')}`;
  const titles = [
    '取引残高報告書','株式等利益剰余金配当金のお知らせ','株式等配当金のお知らせ',
    '割当株式等のお知らせ','割当株券のお知らせ','譲渡益税徴収・還付のお知らせ',
    '外国債券利金・償還金お支払のご案内','外国債券利金・償還金お支払のご案内（兼）支払通知書',
    '取引報告書'
  ];
  for (let i=0; i<actions.length; i++) {
    const button = actions[i];
    const rect = button.getBoundingClientRect();
    if (!rect.width || !rect.height) continue;
    let parent = button.parentElement;
    for (let depth=0; parent && depth<10; parent=parent.parentElement,depth++) {
      const text = (parent.innerText || '').replace(/\s+/g,' ').trim();
      const dates = [...text.matchAll(/20\d{2}\/\d{2}\/\d{2}/g)];
      const date = dates.find(item =>
        item[0] <= today && !/有効期限\s*$/.test(text.slice(Math.max(0,item.index-10),item.index)));
      if (!date || text.length > 1800) continue;
      const title = titles.find(value => text.includes(value));
      if (!title) continue;
      const label = (button.innerText || button.textContent ||
        button.getAttribute('aria-label') || button.getAttribute('title') || '').trim();
      const rank = (label === '+' || /展開|詳細を表示/.test(label) ? 100000 : 0) + rect.right;
      const excerpt = text.slice(0,320);
      const key = date[0]+'|'+title+'|'+excerpt;
      const old = byCard.get(key);
      const top = (typeof window === 'undefined' ? 0 : window.scrollY) + (rect.top || 0);
      if (!old || rank > old.rank) byCard.set(key,{index:i,date:date[0],title,key,rank,top});
      break;
    }
  }
  const items = [...byCard.values()].sort((a,b) => a.top-b.top || a.index-b.index);
  const seen = new Map();
  return items.map(item => {
    const pair=item.date+'|'+item.title;
    const occurrence=seen.get(pair)||0; seen.set(pair,occurrence+1);
    return {index:item.index,date:item.date,title:item.title,key:item.key,top:item.top,occurrence};
  });
}"""

FIND_VIEW = r"""(index) => {
  const controls=[...document.querySelectorAll('button,[role="button"],a')];
  const button=controls[index]; if (!button) return -1;
  let parent=button.parentElement;
  for(let depth=0;parent && depth<10;parent=parent.parentElement,depth++) {
    const text=(parent.innerText || '').replace(/\s+/g,' ').trim();
    if(text.length>2500 || !/20\d{2}\/\d{2}\/\d{2}/.test(text)) continue;
    const views=[...parent.querySelectorAll('button,[role="button"],a')];
    let best=null;
    for(const view of views) {
      if(view===button) continue;
      const rect=view.getBoundingClientRect();
      if(!rect.width || !rect.height) continue;
      const label=(view.innerText || view.textContent || view.getAttribute('aria-label') ||
        view.getAttribute('title') || '').replace(/\s+/g,' ').trim();
      const href=(view.getAttribute('href') || '');
      const score=(/PDFファイル|PDF/.test(label)?10000:0)+
        (/ダウンロード|閲覧|保存/.test(label)?5000:0)+
        (/\.pdf(?:$|\?)/i.test(href)?2000:0)+rect.right;
      if(score>0 && (!best || score>best.score)) best={index:controls.indexOf(view),score};
    }
    if(best) return best.index;
  }
  return -1;
}"""

def _resolve_index(page, item):
    try:
        page.evaluate('(top) => window.scrollTo(0,Math.max(0,top-160))', item.get('top', 0))
        page.wait_for_timeout(250)
    except Exception:
        pass
    current = page.evaluate(DISCOVER)
    exact = next((value for value in current if value['key'] == item['key']), None)
    if exact:
        return exact['index']
    same = [value for value in current
            if value['date'] == item['date'] and value['title'] == item['title']]
    occurrence = item.get('occurrence', 0)
    return same[occurrence]['index'] if occurrence < len(same) else -1

def _retrieve(page, context, item, responses, downloads, popups):
    starts = (len(responses), len(downloads), len(popups))
    index = _resolve_index(page, item)
    if index < 0:
        raise ValueError('一覧内の書類を再発見できませんでした。')
    controls = page.locator('button,[role="button"],a')
    controls.nth(index).scroll_into_view_if_needed(timeout=5000)
    controls.nth(index).click(timeout=7000)
    page.wait_for_timeout(650)
    raw = _wait_pdf(page, responses, downloads, popups, starts, seconds=2)
    if raw is None:
        index = _resolve_index(page, item)
        view = page.evaluate(FIND_VIEW, index)
        if view < 0:
            # Re-resolve because expanding a card may have changed the control list.
            index = _resolve_index(page, item)
            view = page.evaluate(FIND_VIEW, index)
        if view >= 0:
            page.locator('button,[role="button"],a').nth(view).click(timeout=7000)
            raw = _wait_pdf(page, responses, downloads, popups, starts)
    if raw is None:
        raise ValueError('PDF保存ボタンまたはPDF本体を見つけられませんでした。')
    return raw, starts[2]

def _wait_pdf(page, responses, downloads, popups, starts, seconds=12):
    deadline = time.monotonic() + seconds
    while time.monotonic() < deadline:
        raw = take_pdf(responses[starts[0]:], downloads[starts[1]:], popups[starts[2]:])
        if raw is not None:
            return raw
        page.wait_for_timeout(350)
    return None

def discover_all(page, max_scrolls=100):
    """Scan the whole lazy list and deduplicate cards by their stable text."""
    items = {}
    stable_bottom = 0
    last = None
    page.evaluate('window.scrollTo(0,0)')
    page.wait_for_timeout(300)
    for _ in range(max_scrolls):
        for item in page.evaluate(DISCOVER):
            previous = items.get(item['key'])
            if previous is None or item['top'] < previous['top']:
                items[item['key']] = item
        state = page.evaluate("""() => {
          const before=window.scrollY;
          const step=Math.max(450,Math.floor(window.innerHeight*0.80));
          window.scrollBy(0,step);
          return {before,after:window.scrollY,height:document.documentElement.scrollHeight,
            viewport:window.innerHeight};
        }""")
        marker = (state['after'], state['height'], len(items))
        if marker == last or state['after'] + state['viewport'] >= state['height'] - 4:
            stable_bottom += 1
        else:
            stable_bottom = 0
        last = marker
        page.wait_for_timeout(300)
        if stable_bottom >= 3:
            break
    if stable_bottom < 3:raise RuntimeError('一覧の末尾を確認できません。期間を絞って再実行してください。')
    page.evaluate('window.scrollTo(0,0)')
    page.wait_for_timeout(300)
    return sorted(items.values(), key=lambda item: (item['top'], item['date'], item['title']))

def take_pdf(caught_responses, caught_downloads, pages):
    """Accept only genuine PDF bytes from a browser action, never HTML."""
    for download in caught_downloads:
        try:
            path = download.path()
            if path:
                raw = Path(path).read_bytes()
                if raw.startswith(b'%PDF-') and len(raw) <= MAX_PDF_BYTES:
                    return raw
        except Exception:
            pass
    for response in caught_responses:
        try:
            if 'pdf' in response.headers.get('content-type', '').lower():
                raw = response.body()
                if raw.startswith(b'%PDF-') and len(raw) <= MAX_PDF_BYTES:
                    return raw
        except Exception:
            pass
    for page in pages:
        try:
            if page.is_closed() or not page.url.startswith('blob:'):
                continue
            data = page.evaluate("""async () => {
              const blob=await (await fetch(location.href)).blob();
              if(blob.size>25000000) return null;
              return await new Promise((resolve,reject) => {
                const reader=new FileReader();
                reader.onload=()=>resolve(reader.result.split(',')[1]);
                reader.onerror=reject; reader.readAsDataURL(blob);
              });
            }""")
            if data:
                raw = base64.b64decode(data)
                if raw.startswith(b'%PDF-'):
                    return raw
        except Exception:
            pass
    return None
